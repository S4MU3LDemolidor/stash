```swift
import CoreGraphics
import Foundation

/// Recognizes a quick left-right shake in a stream of mouse positions (the shelf's trigger).
///
/// A shake is `reversalsNeeded` changes of horizontal direction within `window` seconds, where
/// every swing covers at least `minimumSwing` points and goes more sideways than up or down.
/// Coming back less than `hysteresis` points is noise, not a reversal.
struct ShakeDetector {
    var minimumSwing: CGFloat
    var window: TimeInterval
    var reversalsNeeded: Int
    var hysteresis: CGFloat

    private var swingStart: CGPoint?
    private var extreme: CGPoint = .zero
    /// -1 left, +1 right, 0 not moving yet.
    private var direction: CGFloat = 0
    private var reversals: [TimeInterval] = []
    private var fired = false

    init(minimumSwing: CGFloat = 30, window: TimeInterval = 0.6, reversalsNeeded: Int = 3, hysteresis: CGFloat = 6) {
        self.minimumSwing = minimumSwing
        self.window = window
        self.reversalsNeeded = reversalsNeeded
        self.hysteresis = hysteresis
    }

    /// Forgets the movement so far (a new drag started, or the mouse button came up).
    mutating func reset() {
        swingStart = nil
        extreme = .zero
        direction = 0
        reversals.removeAll()
        fired = false
    }

    /// Feeds one mouse position. Returns true once per drag, when the movement amounts to a shake.
    mutating func add(_ point: CGPoint, at time: TimeInterval) -> Bool {
        guard let start = swingStart else {
            swingStart = point
            extreme = point
            return false
        }
        if direction == 0 {
            let dx = point.x - start.x
            if abs(dx) >= hysteresis {
                direction = dx > 0 ? 1 : -1
                extreme = point
            }
            return false
        }
        // Still going the same way: the swing gets longer.
        if (point.x - extreme.x) * direction >= 0 {
            extreme = point
            return false
        }
        guard abs(point.x - extreme.x) >= hysteresis else { return false }

        // A reversal: judge the swing that just ended.
        let horizontal = abs(extreme.x - start.x)
        let vertical = abs(extreme.y - start.y)
        if horizontal >= minimumSwing, horizontal > vertical {
            reversals.append(time)
        } else {
            reversals.removeAll()
        }
        reversals.removeAll { time - $0 > window }
        direction = -direction
        swingStart = extreme
        extreme = point
        guard !fired, reversals.count >= reversalsNeeded else { return false }
        fired = true
        return true
    }
}
```

```swift
import AppKit

/// Watches mouse drags in other apps and reports a shake while something is being dragged.
///
/// Global monitors for mouse events don't need the Accessibility permission (only key events do).
/// A drag-and-drop session writes the system drag pasteboard when it starts; selecting text or
/// moving a window doesn't, so the pasteboard's change count tells a real drag from a plain one.
@MainActor
final class DragMonitor {
    /// A shake happened at this point (screen coordinates).
    var onShake: ((NSPoint) -> Void)?
    /// The mouse button came up after a drag with content.
    var onDragEnded: (() -> Void)?

    private var monitors: [Any] = []
    private var detector = ShakeDetector()
    private var dragChangeCount = NSPasteboard(name: .drag).changeCount
    private var dragging = false

    var isRunning: Bool { !monitors.isEmpty }

    func start() {
        guard monitors.isEmpty else { return }
        dragChangeCount = NSPasteboard(name: .drag).changeCount
        if let monitor = NSEvent.addGlobalMonitorForEvents(matching: .leftMouseDragged, handler: { [weak self] event in
            MainActor.assumeIsolated { self?.dragged(event) }
        }) {
            monitors.append(monitor)
        }
        if let monitor = NSEvent.addGlobalMonitorForEvents(matching: .leftMouseUp, handler: { [weak self] _ in
            MainActor.assumeIsolated { self?.mouseUp() }
        }) {
            monitors.append(monitor)
        }
    }

    func stop() {
        monitors.forEach(NSEvent.removeMonitor)
        monitors.removeAll()
        detector.reset()
        dragging = false
    }

    private func dragged(_ event: NSEvent) {
        let count = NSPasteboard(name: .drag).changeCount
        if count != dragChangeCount {
            dragChangeCount = count
            dragging = true
            detector.reset()
        }
        guard dragging else { return }
        let location = NSEvent.mouseLocation
        if detector.add(location, at: event.timestamp) {
            onShake?(location)
        }
    }

    private func mouseUp() {
        if dragging { onDragEnded?() }
        dragging = false
        detector.reset()
    }
}
```

```swift
import Foundation
import UniformTypeIdentifiers

/// Something held on the shelf. Files are held by reference (a bookmark to the original);
/// things with no file of their own are kept as private copies in the shelf's `Items` folder.
struct ShelfItem: Codable, Identifiable, Equatable, Sendable {
    enum Kind: Codable, Equatable, Sendable {
        /// A file or folder somewhere on disk.
        case file(bookmark: Data, path: String, isDirectory: Bool)
        /// A private copy inside `Items/` (path relative to it): an image, a text or a promised file.
        case stored(fileName: String, type: StoredType)
        case link(URL, title: String?)
    }

    enum StoredType: String, Codable, Sendable {
        case image, text, file
    }

    var id: UUID
    var added: Date
    /// What the shelf shows: a file name, the start of a text, a page title or a host.
    var name: String
    var kind: Kind

    init(id: UUID = UUID(), added: Date = Date(), name: String, kind: Kind) {
        self.id = id
        self.added = added
        self.name = name
        self.kind = kind
    }

    /// A reference to a file or folder on disk.
    static func file(at url: URL) throws -> ShelfItem {
        let isDirectory = (try? url.resourceValues(forKeys: [.isDirectoryKey]).isDirectory) ?? url.hasDirectoryPath
        let bookmark = try url.bookmarkData(options: [], includingResourceValuesForKeys: nil, relativeTo: nil)
        return ShelfItem(name: url.lastPathComponent, kind: .file(bookmark: bookmark, path: url.path, isDirectory: isDirectory))
    }

    /// The first line of a text, at most 60 characters (plus an ellipsis).
    static func preview(of text: String) -> String {
        let line = text.trimmingCharacters(in: .whitespacesAndNewlines)
            .split(whereSeparator: \.isNewline).first.map(String.init) ?? ""
        return line.count > 60 ? String(line.prefix(60)) + "\u{2026}" : line
    }

    var isPrivateCopy: Bool {
        if case .stored = kind { return true }
        return false
    }

    var isFolder: Bool {
        if case .file(_, _, let isDirectory) = kind { return isDirectory }
        return false
    }
}
```

```swift
import Foundation
import UniformTypeIdentifiers

/// The shelf's contents and their persistence in `~/Library/Application Support/Stash/Shelf`:
/// `shelf.json` lists the items (newest first) and `Items/` holds the private copies.
/// It never touches the library folder, and never moves or deletes an original file.
@MainActor
final class ShelfStore: ObservableObject {
    @Published private(set) var items: [ShelfItem] = []
    let folder: URL

    var itemsFolder: URL { folder.appending(path: "Items", directoryHint: .isDirectory) }
    private var manifest: URL { folder.appending(path: "shelf.json") }

    static var defaultFolder: URL {
        URL.applicationSupportDirectory.appending(path: "Stash/Shelf", directoryHint: .isDirectory)
    }

    init(folder: URL = ShelfStore.defaultFolder) {
        self.folder = folder
        load()
    }

    // MARK: Changes

    /// Puts items on top of the shelf.
    func add(_ new: [ShelfItem]) {
        guard !new.isEmpty else { return }
        items.insert(contentsOf: new, at: 0)
        save()
    }

    /// Takes items off the shelf and deletes their private copies.
    func remove(_ ids: Set<UUID>) {
        let leaving = items.filter { ids.contains($0.id) }
        guard !leaving.isEmpty else { return }
        items.removeAll { ids.contains($0.id) }
        for item in leaving { deleteCopy(of: item) }
        save()
    }

    func clear() {
        remove(Set(items.map(\.id)))
    }

    /// Whether clearing would lose something that exists only on the shelf.
    var holdsPrivateCopies: Bool { items.contains(where: \.isPrivateCopy) }

    // MARK: Private copies

    /// Writes `data` as a new private copy; returns its name inside `Items/`.
    func storeCopy(_ data: Data, fileExtension: String) throws -> String {
        try FileManager.default.createDirectory(at: itemsFolder, withIntermediateDirectories: true)
        let name = UUID().uuidString + "." + fileExtension
        try data.write(to: itemsFolder.appending(path: name), options: .atomic)
        return name
    }

    /// A new folder inside `Items/` to receive one promised file (keeps its original name).
    func makePromiseFolder() throws -> URL {
        let url = itemsFolder.appending(path: UUID().uuidString, directoryHint: .isDirectory)
        try FileManager.default.createDirectory(at: url, withIntermediateDirectories: true)
        return url
    }

    /// Adds a file that arrived inside a promise folder.
    func addReceivedFile(_ url: URL) {
        let relative = url.deletingLastPathComponent().lastPathComponent + "/" + url.lastPathComponent
        let isImage = UTType(filenameExtension: url.pathExtension)?.conforms(to: .image) ?? false
        add([ShelfItem(name: url.lastPathComponent, kind: .stored(fileName: relative, type: isImage ? .image : .file))])
    }

    // MARK: Resolving

    /// The file an item stands for: the original (following moves and renames) or its private
    /// copy. Nil for links, and for originals that are gone or in the Trash.
    func fileURL(for item: ShelfItem) -> URL? {
        switch item.kind {
        case .link:
            return nil
        case .stored(let fileName, _):
            let url = itemsFolder.appending(path: fileName)
            return FileManager.default.fileExists(atPath: url.path) ? url : nil
        case .file(let bookmark, _, let isDirectory):
            var stale = false
            guard let url = try? URL(resolvingBookmarkData: bookmark, options: [.withoutUI, .withoutMounting], relativeTo: nil, bookmarkDataIsStale: &stale),
                  FileManager.default.fileExists(atPath: url.path),
                  !url.path.contains("/.Trash/")
            else { return nil }
            if stale, let fresh = try? url.bookmarkData(options: [], includingResourceValuesForKeys: nil, relativeTo: nil),
               let index = items.firstIndex(where: { $0.id == item.id }) {
                items[index].kind = .file(bookmark: fresh, path: url.path, isDirectory: isDirectory)
                items[index].name = url.lastPathComponent
                save()
            }
            return url
        }
    }

    func isMissing(_ item: ShelfItem) -> Bool {
        if case .link = item.kind { return false }
        return fileURL(for: item) == nil
    }

    // MARK: Persistence

    private func load() {
        guard let data = try? Data(contentsOf: manifest) else {
            cleanUpCopies()
            return
        }
        do {
            items = try Self.decoder.decode([ShelfItem].self, from: data)
            cleanUpCopies()
        } catch {
            // Keep the broken list for inspection, start empty, and leave Items/ alone this time
            // so no private copy is deleted by mistake.
            let aside = folder.appending(path: "shelf.json.broken")
            try? FileManager.default.removeItem(at: aside)
            try? FileManager.default.moveItem(at: manifest, to: aside)
            items = []
        }
    }

    private func save() {
        do {
            try FileManager.default.createDirectory(at: folder, withIntermediateDirectories: true)
            try Self.encoder.encode(items).write(to: manifest, options: .atomic)
        } catch {
            NSLog("Stash: couldn't save the shelf: \(error.localizedDescription)")
        }
    }

    /// Deletes anything in `Items/` that no item refers to.
    private func cleanUpCopies() {
        let referenced = Set(items.compactMap { item -> String? in
            guard case .stored(let fileName, _) = item.kind else { return nil }
            return String(fileName.split(separator: "/").first ?? "")
        })
        let contents = (try? FileManager.default.contentsOfDirectory(at: itemsFolder, includingPropertiesForKeys: nil)) ?? []
        for url in contents where !referenced.contains(url.lastPathComponent) {
            try? FileManager.default.removeItem(at: url)
        }
```

```swift
import AppKit
import UniformTypeIdentifiers

/// Turns drops into shelf items, and shelf items back into drag data.
@MainActor
enum ShelfPasteboard {
    /// The page title some browsers attach to a dragged link.
    static let urlName = NSPasteboard.PasteboardType("public.url-name")

    /// Types the shelf and the menu bar icon register for.
    static var acceptedTypes: [NSPasteboard.PasteboardType] {
        ClipboardReader.acceptedTypes + NSFilePromiseReceiver.readableDraggedTypes.map { NSPasteboard.PasteboardType($0) }
    }

    /// The items a drop adds right away. Promised files (Photos, Mail) arrive later; see `promises(in:)`.
    static func items(from pasteboard: NSPasteboard, store: ShelfStore) throws -> [ShelfItem] {
        let types = Set(pasteboard.types ?? [])
        guard types.isDisjoint(with: ClipboardReader.privateTypes) else { return [] }

        if let urls = pasteboard.readObjects(forClasses: [NSURL.self], options: [.urlReadingFileURLsOnly: true]) as? [URL], !urls.isEmpty {
            return try urls.map { try ShelfItem.file(at: $0) }
        }
        switch ClipboardReader.content(of: pasteboard) {
        case .image(let data, let fileExtension)?:
            let name = try store.storeCopy(data, fileExtension: fileExtension)
            return [ShelfItem(name: "Image", kind: .stored(fileName: name, type: .image))]
        case .text(let text)?:
            let name = try store.storeCopy(Data(text.utf8), fileExtension: "txt")
            return [ShelfItem(name: ShelfItem.preview(of: text), kind: .stored(fileName: name, type: .text))]
        case .link(let url, _, _)?:
            let raw = pasteboard.string(forType: urlName)?.trimmingCharacters(in: .whitespacesAndNewlines)
            let title = raw?.isEmpty == false ? raw : nil
            let name = title ?? url.host(percentEncoded: false) ?? url.absoluteString
            return [ShelfItem(name: name, kind: .link(url, title: title))]
        case .file(let url)?:
            return [try ShelfItem.file(at: url)]
        case nil:
            return []
        }
    }

    /// File promises in a drop (Photos, Mail attachments, some browsers' images).
    static func promises(in pasteboard: NSPasteboard) -> [NSFilePromiseReceiver] {
        pasteboard.readObjects(forClasses: [NSFilePromiseReceiver.self]) as? [NSFilePromiseReceiver] ?? []
    }

    /// Drag data for items going back out. Missing originals are left out.
    static func writers(for items: [ShelfItem], store: ShelfStore) -> [NSPasteboardWriting] {
        items.compactMap { item -> NSPasteboardWriting? in
            switch item.kind {
            case .link(let url, let title):
                let entry = NSPasteboardItem()
                entry.setString(url.absoluteString, forType: .URL)
                entry.setString(url.absoluteString, forType: .string)
                if let title { entry.setString(title, forType: urlName) }
                return entry
            case .stored(_, let type):
                guard let url = store.fileURL(for: item) else { return nil }
                let entry = NSPasteboardItem()
                entry.setString(url.absoluteString, forType: .fileURL)
                if type == .text, let text = try? String(contentsOf: url, encoding: .utf8) {
                    entry.setString(text, forType: .string)
                }
                if type == .image, let data = try? Data(contentsOf: url),
                   let uti = UTType(filenameExtension: url.pathExtension) {
                    entry.setData(data, forType: NSPasteboard.PasteboardType(uti.identifier))
                }
                return entry
            case .file:
                return store.fileURL(for: item).map { $0 as NSURL }
            }
        }
    }
}
```

```swift
import AppKit
import SwiftUI

/// An invisible AppKit view that starts a real drag session. SwiftUI's `onDrag` can't tell where
/// a drag ended or with which operation, and the shelf needs both to know when an item has left.
struct ShelfDragSource: NSViewRepresentable {
    let writers: () -> [NSPasteboardWriting]
    let image: () -> NSImage?
    let onClick: () -> Void
    /// The operation the drop target chose (empty when cancelled or refused).
    let onEnded: (NSDragOperation) -> Void

    func makeNSView(context: Context) -> DragSourceView {
        let view = DragSourceView()
        update(view)
        return view
    }

    func updateNSView(_ view: DragSourceView, context: Context) {
        update(view)
    }

    private func update(_ view: DragSourceView) {
        view.writers = writers
        view.image = image
        view.onClick = onClick
        view.onEnded = onEnded
    }
}

final class DragSourceView: NSView, NSDraggingSource {
    var writers: () -> [NSPasteboardWriting] = { [] }
    var image: () -> NSImage? = { nil }
    var onClick: () -> Void = {}
    var onEnded: (NSDragOperation) -> Void = { _ in }
    private var mouseDown: NSEvent?

    /// Clicks and drags work even though Stash is never the active app.
    override func acceptsFirstMouse(for event: NSEvent?) -> Bool { true }

    override func mouseDown(with event: NSEvent) {
        mouseDown = event
    }

    override func mouseDragged(with event: NSEvent) {
        guard let down = mouseDown else { return }
        let dx = event.locationInWindow.x - down.locationInWindow.x
        let dy = event.locationInWindow.y - down.locationInWindow.y
        guard dx * dx + dy * dy > 9 else { return }
        mouseDown = nil
        let preview = image() ?? NSWorkspace.shared.icon(for: .data)
        let side = min(bounds.width, bounds.height, 96)
        let frame = NSRect(x: bounds.midX - side / 2, y: bounds.midY - side / 2, width: side, height: side)
        let items = writers().map { writer -> NSDraggingItem in
            let item = NSDraggingItem(pasteboardWriter: writer)
            item.setDraggingFrame(frame, contents: preview)
            return item
        }
        guard !items.isEmpty else { return }
        let session = beginDraggingSession(with: items, event: down, source: self)
        session.animatesToStartingPositionsOnCancelOrFail = true
        session.draggingFormation = .stack
    }

    override func mouseUp(with event: NSEvent) {
        if mouseDown != nil { onClick() }
        mouseDown = nil
    }

    // MARK: NSDraggingSource

    /// The drop target decides (Finder: same disk moves, other disk copies, ⌥ copies). No delete:
    /// the Trash refuses shelf items, so an original can't be trashed by accident.
    func draggingSession(_ session: NSDraggingSession, sourceOperationMaskFor context: NSDraggingContext) -> NSDragOperation {
        context == .outsideApplication ? [.copy, .move, .link, .generic] : []
    }

    func draggingSession(_ session: NSDraggingSession, endedAt screenPoint: NSPoint, operation: NSDragOperation) {
        onEnded(operation)
    }
}

/// The strip at the top of the shelf: dragging it moves the window.
struct ShelfMoveHandle: NSViewRepresentable {
    func makeNSView(context: Context) -> MoveHandleView { MoveHandleView() }
    func updateNSView(_ view: MoveHandleView, context: Context) {}
}

final class MoveHandleView: NSView {
    override func acceptsFirstMouse(for event: NSEvent?) -> Bool { true }

    override func mouseDown(with event: NSEvent) {
        window?.performDrag(with: event)
    }
}
```

```swift
import AppKit

/// The shelf's window: borderless, non-activating, floating on every Space (like `StashPanel`),
/// but it never takes keyboard focus; everything happens with the mouse.
final class ShelfPanel: NSPanel {
    init() {
        super.init(
            contentRect: NSRect(x: 0, y: 0, width: Metrics.shelfWidth, height: Metrics.shelfCollapsedHeight),
            styleMask: [.borderless, .nonactivatingPanel, .fullSizeContentView],
            backing: .buffered,
            defer: true
        )
        isFloatingPanel = true
        level = .floating
        collectionBehavior = [.canJoinAllSpaces, .fullScreenAuxiliary, .ignoresCycle]
        hidesOnDeactivate = false
        isReleasedWhenClosed = false
        animationBehavior = .none
        isOpaque = false
        backgroundColor = .clear
        hasShadow = false
        isMovableByWindowBackground = false
    }

    override var canBecomeKey: Bool { false }
    override var canBecomeMain: Bool { false }
}

/// The shelf's glass, which also accepts drops.
final class ShelfContentView: NSVisualEffectView {
    var onTargetedChange: ((Bool) -> Void)?
    var onDrop: ((NSPasteboard) -> Bool)?

    override init(frame: NSRect) {
        super.init(frame: frame)
        registerForDraggedTypes(ShelfPasteboard.acceptedTypes)
    }

    @available(*, unavailable)
    required init?(coder: NSCoder) {
        fatalError("init(coder:) is not used")
    }

    override func draggingEntered(_ sender: NSDraggingInfo) -> NSDragOperation {
        // Our own drags (an item being dragged out) don't drop back in.
        guard sender.draggingSource == nil || !(sender.draggingSource is DragSourceView),
              sender.draggingPasteboard.availableType(from: ShelfPasteboard.acceptedTypes) != nil
        else { return [] }
        onTargetedChange?(true)
        return .copy
    }

    override func draggingExited(_ sender: NSDraggingInfo?) {
        onTargetedChange?(false)
    }

    override func draggingEnded(_ sender: NSDraggingInfo) {
        onTargetedChange?(false)
    }

    override func performDragOperation(_ sender: NSDraggingInfo) -> Bool {
        onTargetedChange?(false)
        return onDrop?(sender.draggingPasteboard) ?? false
    }
}
```

```swift
import AppKit
import Combine
import SwiftUI

/// Shows, sizes and positions the shelf, takes drops, and takes items off once they've been
/// dragged out (unless ⌥ was held). The shelf stays on screen while it holds anything.
@MainActor
final class ShelfController: NSObject, NSWindowDelegate {
    let store: ShelfStore
    private let state = ShelfViewState()
    private let thumbnails = ShelfThumbnails()
    private let panel = ShelfPanel()
    private let shadowWindow = PanelShadowWindow()
    private var cancellables = Set<AnyCancellable>()
    private var isShown = false
    private var hideWork: DispatchWorkItem?

    init(store: ShelfStore) {
        self.store = store
        super.init()

        let content = ShelfContentView(frame: NSRect(origin: .zero, size: panel.frame.size))
        Glass.configure(content, cornerRadius: Metrics.panelCornerRadius)
        content.onTargetedChange = { [weak self] in self?.state.dropTargeted = $0 }
        content.onDrop = { [weak self] in self?.accept($0) ?? false }
        let hosting = NSHostingView(rootView: ShelfView(store: store, state: state, thumbnails: thumbnails, actions: actions))
        hosting.frame = content.bounds
        hosting.autoresizingMask = [.width, .height]
        content.addSubview(hosting)
        panel.contentView = content
        panel.delegate = self
        panel.setAccessibilityLabel("Shelf")

        shadowWindow.level = panel.level
        shadowWindow.collectionBehavior = panel.collectionBehavior

        store.$items
            .receive(on: RunLoop.main)
            .sink { [weak self] items in self?.itemsChanged(items) }
            .store(in: &cancellables)
    }

    // MARK: Showing

    /// Opens the shelf centered on `point` (screen coordinates), or moves it there.
    func show(at point: NSPoint) {
        hideWork?.cancel()
        let size = targetSize
        place(NSRect(x: point.x - size.width / 2, y: point.y - size.height / 2, width: size.width, height: size.height))
        present()
    }

    /// At launch: a shelf that still holds items comes back where it was.
    func restore() {
        guard !store.items.isEmpty else { return }
        let size = targetSize
        if let origin = Preferences.shelfOrigin {
            place(NSRect(origin: origin, size: size))
            present()
        } else if let screen = NSScreen.main {
            show(at: NSPoint(x: screen.visibleFrame.maxX - size.width, y: screen.visibleFrame.midY))
        }
    }

    /// A drag ended somewhere. An empty shelf (shaken open, then not used) fades out.
    func dragEnded() {
        scheduleHideIfEmpty()
    }

    /// Takes a drop (on the shelf or on the menu bar icon). Returns whether anything was added.
    @discardableResult
    func accept(_ pasteboard: NSPasteboard) -> Bool {
        let promises = ShelfPasteboard.promises(in: pasteboard)
        if !promises.isEmpty {
            receive(promises)
            return true
        }
        do {
            let items = try ShelfPasteboard.items(from: pasteboard, store: store)
            guard !items.isEmpty else { return false }
            store.add(items)
            return true
        } catch {
            HUDController.shared.show(.shelfFailed)
            return false
        }
    }

    // MARK: Actions from the view

    private var actions: ShelfActions {
        ShelfActions(
            writers: { [weak self] items in
                guard let self else { return [] }
                return ShelfPasteboard.writers(for: items, store: self.store)
            },
            dragEnded: { [weak self] items, operation in self?.dragged(items, operation: operation) },
            toggleExpanded: { [weak self] in self?.toggleExpanded() },
            clear: { [weak self] in self?.clear() },
            remove: { [weak self] item in self?.store.remove([item.id]) },
            fileURL: { [weak self] item in self?.store.fileURL(for: item) },
            isMissing: { [weak self] item in self?.store.isMissing(item) ?? true }
        )
    }

    private func dragged(_ items: [ShelfItem], operation: NSDragOperation) {
        // Cancelled or refused: everything stays. ⌥ held: the drop made a copy, keep ours too.
        guard !operation.isEmpty, !NSEvent.modifierFlags.contains(.option) else { return }
        store.remove(Set(items.map(\.id)))
    }

    private func toggleExpanded() {
        state.expanded.toggle()
        resize(animated: true)
    }

    private func clear() {
        if store.holdsPrivateCopies {
            let alert = NSAlert()
            alert.messageText = "Clear the shelf?"
            alert.informativeText = "Some items exist only on the shelf (things dragged from a browser or another app) and will be deleted. Your files stay where they are."
            alert.addButton(withTitle: "Clear")
            alert.addButton(withTitle: "Cancel")
            NSApp.activate()
            guard alert.runModal() == .alertFirstButtonReturn else { return }
        }
        store.clear()
    }

    // MARK: Promised files

    private func receive(_ promises: [NSFilePromiseReceiver]) {
        let queue = OperationQueue()
        queue.qualityOfService = .userInitiated
        for receiver in promises {
            let folder: URL
            do {
                folder = try store.makePromiseFolder()
            } catch {
                HUDController.shared.show(.shelfFailed)
                continue
            }
            receiver.receivePromisedFiles(atDestination: folder, options: [:], operationQueue: queue) { [weak self] url, error in
                DispatchQueue.main.async {
                    MainActor.assumeIsolated {
                        guard let self else { return }
                        if error != nil {
                            try? FileManager.default.removeItem(at: folder)
                            HUDController.shared.show(.shelfFailed)
                        } else {
                            self.store.addReceivedFile(url)
                        }
                    }
                }
            }
        }
    }

    // MARK: Layout

    private func itemsChanged(_ items: [ShelfItem]) {
        thumbnails.keep(only: Set(items.map(\.id)))
        for item in items { thumbnails.load(item, url: store.fileURL(for: item)) }
        if items.isEmpty {
            state.expanded = false
            hide()
        } else if isShown {
            resize(animated: true)
        }
    }

    private var targetSize: NSSize {
        guard state.expanded else { return NSSize(width: Metrics.shelfWidth, height: Metrics.shelfCollapsedHeight) }
        let rows = CGFloat(min(max(store.items.count, 1), Metrics.shelfMaxVisibleRows))
        return NSSize(width: Metrics.shelfExpandedWidth, height: Metrics.shelfHeaderHeight + rows * Metrics.shelfRowHeight + Metrics.contentInset)
    }

    /// Keeps the top-left corner where it is.
    private func resize(animated: Bool) {
        let old = panel.frame
        let size = targetSize
        let frame = clamped(NSRect(x: old.minX, y: old.maxY - size.height, width: size.width, height: size.height))
        if animated, isShown, !NSWorkspace.shared.accessibilityDisplayShouldReduceMotion {
            NSAnimationContext.runAnimationGroup { context in
                context.duration = Motion.modeSwitch
                panel.animator().setFrame(frame, display: true)
            }
            shadowWindow.layout(around: frame)
        } else {
            panel.setFrame(frame, display: true)
            shadowWindow.layout(around: frame)
        }
    }

    private func place(_ frame: NSRect) {
        let frame = clamped(frame)
        panel.setFrame(frame, display: true)
        shadowWindow.layout(around: frame)
        Preferences.shelfOrigin = frame.origin
    }

    /// Pulls the frame onto the screen it's mostly on (or the main screen if that one is gone).
    private func clamped(_ frame: NSRect) -> NSRect {
        let screen = NSScreen.screens.first { $0.frame.contains(NSPoint(x: frame.midX, y: frame.midY)) } ?? NSScreen.main
        guard let visible = screen?.visibleFrame else { return frame }
        var result = frame
        result.origin.x = min(max(result.minX, visible.minX + 8), visible.maxX - result.width - 8)
        result.origin.y = min(max(result.minY, visible.minY + 8), visible.maxY - result.height - 8)
        return result
    }

    private func present() {
        guard !isShown else { return }
        isShown = true
        let reduceMotion = NSWorkspace.shared.accessibilityDisplayShouldReduceMotion
        let final = panel.frame
        panel.alphaValue = 0
        shadowWindow.alphaValue = 0
        if !reduceMotion {
            panel.setFrame(final.insetBy(dx: final.width * 0.06, dy: final.height * 0.06), display: false)
        }
        panel.orderFrontRegardless()
        panel.addChildWindow(shadowWindow, ordered: .below)
        NSAnimationContext.runAnimationGroup { context in
            context.duration = Motion.shelfAppear
            context.timingFunction = CAMediaTimingFunction(name: .easeOut)
            panel.animator().alphaValue = 1
            shadowWindow.animator().alphaValue = 1
            if !reduceMotion { panel.animator().setFrame(final, display: true) }
        }
        shadowWindow.layout(around: final)
    }

    private func hide() {
        guard isShown else { return }
        isShown = false
        NSAnimationContext.runAnimationGroup { context in
            context.duration = Motion.hudFadeOut
            panel.animator().alphaValue = 0
            shadowWindow.animator().alphaValue = 0
        } completionHandler: { [weak self] in
            MainActor.assumeIsolated {
                guard let self, !self.isShown else { return }
                self.panel.removeChildWindow(self.shadowWindow)
                self.shadowWindow.orderOut(nil)
                self.panel.orderOut(nil)
            }
        }
    }

    private func scheduleHideIfEmpty() {
        hideWork?.cancel()
        let work = DispatchWorkItem { [weak self] in
            MainActor.assumeIsolated {
                guard let self, self.store.items.isEmpty else { return }
                self.hide()
            }
        }
        hideWork = work
        DispatchQueue.main.asyncAfter(deadline: .now() + Motion.shelfEmptyGrace, execute: work)
    }

    // MARK: NSWindowDelegate

    func windowDidMove(_ notification: Notification) {
        shadowWindow.layout(around: panel.frame)
        Preferences.shelfOrigin = panel.frame.origin
    }

    // MARK: Debug

    #if DEBUG
    var isExpanded: Bool { state.expanded }
    func setExpanded(_ expanded: Bool) {
        guard state.expanded != expanded else { return }
        toggleExpanded()
    }
    var windowNumber: Int { panel.windowNumber }
    #endif
}
```
